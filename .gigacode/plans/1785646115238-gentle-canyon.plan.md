# План миграции на SQLite + Prisma

## Цели
- Заменить mock-данные на реальную базу данных SQLite
- Настроить Prisma ORM для типобезопасной работы
- Подготовить schema для всех сущностей проекта
- Создать Server Actions для CRUD-операций

## Этап 1: Установка и настройка

### 1.1 Установка зависимостей
```bash
npm install prisma --save-dev
npm install @prisma/client
```

### 1.2 Инициализация Prisma
```bash
npx prisma init
```

### 1.3 Обновление `prisma/schema.prisma`
- Настройка provider = "sqlite"
- Указание пути к БД: `./dev.db`
- Настройка strict mode и nativeTypes

## Этап 2: Схема БД (Prisma Schema)

### 2.1 Сущности

```prisma
// Пользователи (подготовка для мультипользовательского режима)
model User {
  id        Int      @id @default(autoincrement())
  email     String   @unique
  name      String?
  role      String   @default("admin") // admin, manager, worker
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
}

// Проекты/Объекты
model Project {
  id            Int      @id @default(autoincrement())
  name          String
  area          String   // площадь в м²
  address       String
  type          String   // дом, баня, туалет, хозблок, веранда
  cost          Int      // стоимость в рублях
  deadline      DateTime // дата сдачи
  complexity    String   // легкий, средний, сложный
  status        String   // создан, в работе, завершен
  code          String   @unique
  prepayment    Int?     // сумма предоплаты
  prepaymentDate DateTime?
  brigadeId     Int?
  brigade       Brigade? @relation(fields: [brigadeId], references: [id])
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt
}

// Бригады
model Brigade {
  id        Int      @id @default(autoincrement())
  name      String
  leaderId  Int      // ID прораба
  memberIds String   // JSON-строка с массивом ID участников [2, 4]
  skills    String   // JSON-строка с навыками ["сварщик", "монтажник"]
  projects  Project[]
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
}

// Сотрудники
model Employee {
  id            Int      @id @default(autoincrement())
  fullName      String
  birthDate     DateTime
  phone         String
  address       String
  hireDate      DateTime
  workplace     String
  paymentType   String
  employmentType String
  skills        String   // JSON-строка ["сварщик", "плотник"]
  brigadeId     Int?
  brigade       Brigade? @relation(fields: [brigadeId], references: [id])
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt
}

// Товары на складе
model WarehouseItem {
  id          Int      @id @default(autoincrement())
  name        String
  category    String   // кирпич, цемент, дерево, металл, инструменты и т.д.
  quantity    Int
  unit        String   // шт, м³, кг, м²
  location    String   // Склад А, секция 1
  lastUpdate  DateTime
  status      String   // достаточно, мало, критически мало, нет в наличии
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
}

// Перемещения товаров (движение склада)
model WarehouseMovement {
  id          Int         @id @default(autoincrement())
  itemId      Int
  item        WarehouseItem @relation(fields: [itemId], references: [id])
  type        String      // income, expense
  quantity    Int
  date        DateTime
  comment     String?
  createdAt   DateTime    @default(now())
}

// Смета материалов (привязана к проекту)
model MaterialEstimate {
  id        Int    @id @default(autoincrement())
  projectId Int
  project   Project @relation(fields: [projectId], references: [id])
  name      String // название материала
  quantity  String // количество с единицами
  cost      Int    // стоимость в рублях
  createdAt DateTime @default(now())
}

// Смета выполненных работ (привязана к проекту)
model CompletedWork {
  id        Int    @id @default(autoincrement())
  projectId Int
  project   Project @relation(fields: [projectId], references: [id])
  name      String // название работы
  quantity  String // количество с единицами
  cost      Int    // стоимость в рублях
  createdAt DateTime @default(now())
}

// Категории оплаты труда
model PaymentCategory {
  id        Int    @id @default(autoincrement())
  name      String // Цех, Монтажники, Премия цеха и т.д.
  percentage Int   // процент
  createdAt DateTime @default(now())
}

// Финансовые фонды
model Fund {
  id          Int      @id @default(autoincrement())
  name        String
  description String?
  balance     Int      @default(0)
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
}

// Транзакции фондов
model FundTransaction {
  id          Int      @id @default(autoincrement())
  fundId      Int
  fund        Fund     @relation(fields: [fundId], references: [id])
  amount      Int
  type        String   // income, expense
  description String?
  date        DateTime
  createdAt   DateTime @default(now())
}

// Отчеты по проектам
model ProjectReport {
  id          Int      @id @default(autoincrement())
  projectId   Int
  project     Project  @relation(fields: [projectId], references: [id])
  date        DateTime
  periodFrom  DateTime
  periodTo    DateTime
  comment     String?
  createdAt   DateTime @default(now())
}

// Служебные записи сотрудников (сдельная оплата)
model EmployeeWorkReport {
  id          Int      @id @default(autoincrement())
  employeeId  Int
  projectId   Int
  project     Project  @relation(fields: [projectId], references: [id])
  workType    String
  quantity    Float
  rate        Float    // ставка за единицу
  amount      Float    // итоговая сумма
  date        DateTime
  comment     String?
  createdAt   DateTime @default(now())
}

// Финансовые отчеты
model FinanceReport {
  id          Int      @id @default(autoincrement())
  projectId   Int?
  project     Project? @relation(fields: [projectId], references: [id])
  date        DateTime
  totalIncome Int
  totalExpense Int
  profit      Int
  comment     String?
  createdAt   DateTime @default(now())
}

// Уведомления (связаны с проектами и складом)
model Notification {
  id        Int      @id @default(autoincrement())
  type      String   // overdue, low-stock, out-of-stock
  message   String
  priority  String   // high, medium
  link      String?
  date      DateTime
  isRead    Boolean  @default(false)
  isArchived Boolean @default(false)
  projectId Int?
  itemId    Int?
  item      WarehouseItem? @relation(fields: [itemId], references: [id])
  createdAt DateTime @default(now())
}
```

### 4.2 Пример `projectActions.ts`
```typescript
'use server';

import { prisma } from '@/lib/prisma';
import { revalidatePath } from 'next/cache';

export async function getProjects() {
  return await prisma.project.findMany({
    include: { brigade: true },
    orderBy: { createdAt: 'desc' },
  });
}

export async function getProjectById(id: number) {
  return await prisma.project.findUnique({
    where: { id },
    include: {
      brigade: true,
      materials: true,
      completedWorks: true,
    },
  });
}

export async function createProject(data: any) {
  return await prisma.project.create({ data });
}

export async function updateProject(id: number, data: any) {
  return await prisma.project.update({ where: { id }, data });
}

export async function deleteProject(id: number) {
  return await prisma.project.delete({ where: { id } });
}
```

## Этап 5: Настройка доступа к БД

### 5.1 Файл `lib/prisma.ts`
```typescript
import { PrismaClient } from '@prisma/client';

const globalForPrisma = globalThis as unknown as { prisma: PrismaClient };

export const prisma = globalForPrisma.prisma || new PrismaClient();

if (process.env.NODE_ENV !== 'production') globalForPrisma.prisma = prisma;
```

### 5.2 Обновление `.env`
```env
DATABASE_URL="file:./dev.db"
```

### 5.3 Обновление `prisma/schema.prisma` для .env
```prisma
datasource db {
  provider = "sqlite"
  url      = env("DATABASE_URL")
}
```
